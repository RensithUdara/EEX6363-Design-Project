class MathUtils {
    public constructor ();
    public func factorial (n: integer) => integer;
    public func fibonacci (n: integer) => integer;
};

implement MathUtils {
    constructor () {
        local temp : integer;
        temp := 0;
    }
    
    func factorial (n: integer) => integer {
        if (n <= 1) then {
            return (1);
        } else {
            return (n * factorial(n - 1));
        }
    }

    func fibonacci (n: integer) => integer {
        if (n <= 1) then {
            return (n);
        } else {
            return (fibonacci(n - 1) + fibonacci(n - 2));
        }
    }
};

class DataProcessor {
    public constructor ();
    public func processData (size: integer) => integer;
};

implement DataProcessor {
    constructor () {
        local temp : integer;
        temp := 100;
    }

    func processData (size: integer) => integer {
        local i: integer;
        local sum: integer;
        sum := 0;
        i := 0;
        
        while (i < size) do {
            sum := sum + i;
            i := i + 1;
        }
        return (sum);
    }
};
