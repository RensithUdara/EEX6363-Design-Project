class Calculator {
    public x: integer;
    public y: integer;
    
    public constructor ();
    
    public func add () => integer;
    public func multiply () => integer;
};

implement Calculator {
    constructor () {
        x := 0;
        y := 0;
    }
    
    func add () => integer {
        return (x + y);
    }
    
    func multiply () => integer {
        return (x * y);
    }
};
