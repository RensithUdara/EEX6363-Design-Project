function main() -> integer {
    local x: integer;
    local y: integer;
    local result: integer;
    
    // Simple arithmetic
    x = 10;
    y = 5;
    result = x + y * 2;
    
    // Conditional statement
    if (result > 15) then {
        result = result - 5;
    } else {
        result = result + 5;
    }
    
    // Function call
    result = factorial(3);
    
    return result;
}

function factorial(n: integer) -> integer {
    local result: integer;
    
    if (n <= 1) then {
        return 1;
    } else {
        result = n * factorial(n - 1);
        return result;
    }
}

function fibonacci(n: integer) -> integer {
    local a: integer;
    local b: integer;
    local temp: integer;
    local i: integer;
    
    if (n <= 1) then {
        return n;
    }
    
    a = 0;
    b = 1;
    i = 2;
    
    while (i <= n) {
        temp = a + b;
        a = b;
        b = temp;
        i = i + 1;
    }
    
    return b;
}
